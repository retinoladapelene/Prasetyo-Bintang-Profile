// Generate a <style> tag that overrides all Tailwind purple-* with theme color
export function ThemeStyleInjector({ activeThemeId }: { activeThemeId: string }) {
    const theme = themeColors.find(t => t.id === activeThemeId) || themeColors[0];

    // Map Tailwind purple shades to theme color with opacity variants
    const hexToRgb = (hex: string) => {
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return `${r} ${g} ${b}`;
    };

    const rgb = hexToRgb(theme.primary);
    const dotRgb = hexToRgb(theme.dot);
    const lightRgb = hexToRgb(theme.light);

    // CSS that overrides every Tailwind purple variant globally
    const css = `
        :root {
            --theme-primary: ${theme.primary};
            --theme-primary-rgb: ${rgb};
            --theme-dot: ${theme.dot};
            --theme-dot-rgb: ${dotRgb};
            --theme-light: ${theme.light};
            --theme-light-rgb: ${lightRgb};
        }
        /* Override all Tailwind purple classes */
        .text-purple-50  { color: rgb(var(--theme-light-rgb) / 0.3) !important; }
        .text-purple-100 { color: rgb(var(--theme-light-rgb)) !important; }
        .text-purple-200 { color: rgb(var(--theme-dot-rgb) / 0.5) !important; }

        .text-purple-600 { color: rgb(var(--theme-primary-rgb)) !important; }
        .text-purple-700 { color: rgb(var(--theme-primary-rgb)) !important; }
        .text-purple-800 { color: rgb(var(--theme-primary-rgb)) !important; }
        .text-purple-900 { color: rgb(var(--theme-primary-rgb)) !important; }

        /* Pseudo-elements and variants */
        .first-letter\\:text-purple-600::first-letter { color: rgb(var(--theme-primary-rgb)) !important; }

        .bg-purple-50  { background-color: rgb(var(--theme-light-rgb) / 0.4) !important; }

function SplitReveal({ text, delay = 0, className = "" }: { text: string; delay?: number; className?: string }) {
    const ref = useRef<HTMLSpanElement>(null);
    const inView = useInView(ref, { once: true, margin: "-60px" });
    return (
        <span ref={ref} className={`inline-block overflow-hidden ${className}`} aria-label={text}>
            <span className="inline-flex flex-wrap">
                {text.split("").map((char, i) => (
                    <motion.span key={i} initial={{ y: "110%", opacity: 0 }}
                        animate={inView ? { y: "0%", opacity: 1 } : {}}
                        transition={{ duration: 0.65, delay: delay + i * 0.028, ease: [0.22, 1, 0.36, 1] }}
                        className="inline-block" style={{ whiteSpace: char === " " ? "pre" : "normal" }}>
                        {char}
                    </motion.span>
                ))}
            </span>
        </span>
    );
}

function SkillMarquee({ skills, theme }: { skills: string[]; theme: typeof themeColors[0] }) {

                        {skill}
                    </span>
                ))}
            </div>
        </div>
    );
}

            </motion.div>
            <div className={`hidden md:block absolute top-1/2 -translate-y-1/2 h-px w-6 pointer-events-none opacity-30 ${side === "left" ? "-right-6" : "-left-6"}`}

// Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡
// PALETTE 3D CANVAS
// Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡Î“Ã¶Ã‡
function PaletteCanvas({ activeThemeId, onColorClick, isActive }: { activeThemeId: string; onColorClick: (id: string) => void; isActive: boolean }) {
    const canvasRef = useRef<HTMLCanvasElement>(null);
    const animRef = useRef<number | null>(null);
    const activeThemeIdRef = useRef(activeThemeId);
    const isActiveRef = useRef(isActive);
    const threeRef = useRef<any>(null);
    // Drag state for 360 rotation
    const dragRef = useRef({ isDragging: false, prevX: 0, prevY: 0, rotY: 0.3, rotX: -0.4, velocityY: 0, velocityX: 0 });
    const onColorClickRef = useRef(onColorClick);

    useEffect(() => { activeThemeIdRef.current = activeThemeId; }, [activeThemeId]);
    useEffect(() => { onColorClickRef.current = onColorClick; }, [onColorClick]);
    useEffect(() => {
        isActiveRef.current = isActive;
        if (isActive && canvasRef.current && (canvasRef.current as any)._startAnim) {
            (canvasRef.current as any)._startAnim();
        }
    }, [isActive]);

    const handleClick = useCallback((e: MouseEvent) => {
        const canvas = canvasRef.current;
        const r = threeRef.current;
        if (!canvas || !r) return;
        // Only fire click if we weren't dragging
        if (Math.abs(dragRef.current.velocityY) > 0.002 || Math.abs(dragRef.current.velocityX) > 0.002) return;
        const { THREE, camera, scene } = r;
        const rect = canvas.getBoundingClientRect();
        const mouse = new THREE.Vector2(
            ((e.clientX - rect.left) / rect.width) * 2 - 1,
            -((e.clientY - rect.top) / rect.height) * 2 + 1
        );
        const raycaster = new THREE.Raycaster();
        raycaster.setFromCamera(mouse, camera);
        const hits = raycaster.intersectObjects(scene.children, true);
        if (hits.length > 0) {
            // Check if we hit a stone hitbox on the model first
            const stoneHit = hits.find((h: any) => h.object.userData && h.object.userData.isStoneHitbox);
            if (stoneHit) {
                onColorClickRef.current(stoneHit.object.userData.themeId);
            }
        }
    }, []);

    useEffect(() => {
        const canvas = canvasRef.current;
        if (!canvas) return;
        let destroyed = false;
        let isVisible = false;
        const isVisibleRef = { current: false };
        const observer = new IntersectionObserver(([entry]) => {
            isVisible = entry.isIntersecting;
            isVisibleRef.current = isVisible;
            if (isVisible && isActiveRef.current && canvasRef.current && (canvasRef.current as any)._startAnim) {
                (canvasRef.current as any)._startAnim();
            }
        }, { threshold: 0.05 });
        observer.observe(canvas);

        const load = async () => {
            // @ts-ignore
            const THREE = await import("three");
            if (destroyed) return;

            const isMobile = window.innerWidth < 768;
            const renderer = new THREE.WebGLRenderer({ canvas, antialias: !isMobile, alpha: true, powerPreference: "high-performance" });
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, isMobile ? 1 : 1.5));
            renderer.setSize(canvas.clientWidth, canvas.clientHeight);
            renderer.setClearColor(0x000000, 0);
            renderer.toneMapping = THREE.ACESFilmicToneMapping;
            renderer.toneMappingExposure = 1.2;

            // Enable PBR sRGB Color Space Encoding
            if ("SRGBColorSpace" in THREE) {
                renderer.outputColorSpace = (THREE as any).SRGBColorSpace;
            }

            const scene = new THREE.Scene();
            const camera = new THREE.PerspectiveCamera(36, canvas.clientWidth / canvas.clientHeight, 0.1, 100);
            camera.position.set(0, 0, 8.5);
            camera.lookAt(0, 0, 0);

            // Studio PBR lighting setup
            scene.add(new THREE.AmbientLight(0xffffff, 1.5));
            const key = new THREE.DirectionalLight(0xffffff, 3.5); key.position.set(5, 8, 6); scene.add(key);
            const fill = new THREE.DirectionalLight(0xffffff, 1.5); fill.position.set(-5, 2, 4); scene.add(fill);
            const back = new THREE.DirectionalLight(0xffffff, 1.0); back.position.set(0, -5, -4); scene.add(back);
            const point = new THREE.PointLight(0xffffff, 2.0, 20); point.position.set(0, 2, 6); scene.add(point);

            const paletteGroup = new THREE.Group();

            // Î“Ã¶Ã‡Î“Ã¶Ã‡ TECH CORE BODY (Infinity Gauntlet GLB) Î“Ã¶Ã‡Î“Ã¶Ã‡
            const coreGrp = new THREE.Group();
            paletteGroup.add(coreGrp);

            // Dynamically import GLTFLoader
            // @ts-ignore
            import("three/examples/jsm/loaders/GLTFLoader.js").then(({ GLTFLoader }) => {
                const loader = new GLTFLoader();
                loader.load('/models/ultro_pbr.glb', (gltf: any) => {
                    const gauntlet = gltf.scene;

                    // Normalize scale to fit the scene perfectly
                    const box = new THREE.Box3().setFromObject(gauntlet);
                    const size = box.getSize(new THREE.Vector3());
                    const maxDim = Math.max(size.x, size.y, size.z);

                    // Increase size dramatically to zoom in (Half-body shot)
                    const desiredSize = 17.0;
                    gauntlet.scale.setScalar(desiredSize / maxDim);

                    // Re-calculate box after scaling to center it exactly at (0,0,0)
                    const scaledBox = new THREE.Box3().setFromObject(gauntlet);
                    const center = scaledBox.getCenter(new THREE.Vector3());
                    gauntlet.position.sub(center);

                    // Shift the model DOWN so the legs are hidden and chest/head is centered
                    gauntlet.position.y -= 6.0;

                    // Preserve native PBR materials & textures
                    gauntlet.traverse((child: any) => {
                        if (child.isMesh && child.material) {
                            child.material.needsUpdate = true;
                            if (child.material.map && THREE.SRGBColorSpace) {
                                child.material.map.colorSpace = THREE.SRGBColorSpace;
                            }
                        }
                    });

                    // Add invisible hitboxes for the infinity stones
                    const hitGeo = new THREE.SphereGeometry(0.08, 8, 8); // small sphere
                    const hitMat = new THREE.MeshBasicMaterial({ color: 0xff0000, transparent: true, opacity: 0 }); // completely invisible

                    const stoneData = [
                        { id: "gold", pos: { x: 0.003, y: 2.046, z: 0.108 } },
                        { id: "emerald", pos: { x: 0.005, y: 1.555, z: 0.208 } },
                        { id: "ruby", pos: { x: -0.041, y: 1.696, z: 0.176 } },
                        { id: "sapphire", pos: { x: 0.108, y: 1.673, z: 0.179 } },
                        { id: "alexandrite", pos: { x: -0.096, y: 1.679, z: 0.178 } },
                        { id: "amethyst", pos: { x: 0.053, y: 1.696, z: 0.176 } }
                    ];

                    stoneData.forEach(sd => {
                        const mesh = new THREE.Mesh(hitGeo, hitMat);
                        mesh.position.set(sd.pos.x, sd.pos.y, sd.pos.z);
                        mesh.userData = { themeId: sd.id, isStoneHitbox: true };
                        gauntlet.add(mesh);
                    });

                    // Expose to window for our temporary coordinate picker
                    // @ts-ignore
                    window.debugGauntlet = gauntlet;

                    // Wrap in another group to apply rotation properly
                    const wrapper = new THREE.Group();
                    wrapper.add(gauntlet);
                    coreGrp.add(wrapper);
                }, undefined, (error: any) => {
                    console.error("Error loading GLB model:", error);
                });
            });

            // Î“Ã¶Ã‡Î“Ã¶Ã‡ SATELLITE NODES REMOVED Î“Ã¶Ã‡Î“Ã¶Ã‡
            // The floating crystals are replaced by the chest stones on the Infinity Ultron model.
            const paintMeshes: any[] = [];

            // Î“Ã¶Ã‡Î“Ã¶Ã‡ Scale & position Î“Ã¶Ã‡Î“Ã¶Ã‡
            paletteGroup.scale.setScalar(0.75);
            scene.add(paletteGroup);

            // Apply initial rotation from dragRef
            paletteGroup.rotation.y = dragRef.current.rotY;
            paletteGroup.rotation.x = dragRef.current.rotX;

            threeRef.current = { THREE, camera, scene, renderer };

            // Î“Ã¶Ã‡Î“Ã¶Ã‡ ANIMATION LOOP Î“Ã¶Ã‡Î“Ã¶Ã‡
            const pIsAnimatingRef = { current: false };
            (canvas as any)._startAnim = () => {
                if (!pIsAnimatingRef.current) animate();
            };

            let t = 0;
            const animate = () => {
                if (destroyed) return;

                if (!isVisibleRef.current || !isActiveRef.current) {
                    pIsAnimatingRef.current = false;
                    return;
                }

                pIsAnimatingRef.current = true;
                animRef.current = requestAnimationFrame(animate);

                t += 0.008;

                const dr = dragRef.current;

                if (!dr.isDragging) {
                    // Apply inertia when not dragging
                    dr.velocityY *= 0.95;
                    dr.velocityX *= 0.95;
                    dr.rotY += dr.velocityY;
                    dr.rotX += dr.velocityX;
                    // Clamp vertical rotation to avoid flipping
                    dr.rotX = Math.max(-1.2, Math.min(0.6, dr.rotX));
                    // Add subtle idle float
                    paletteGroup.rotation.y = dr.rotY + Math.sin(t * 0.25) * 0.02;
                    paletteGroup.rotation.x = dr.rotX + Math.cos(t * 0.2) * 0.01;
                } else {
                    paletteGroup.rotation.y = dr.rotY;
                    paletteGroup.rotation.x = dr.rotX;
                }

                // Active node animations removed

                renderer.render(scene, camera);
            };
            animate();

            // Î“Ã¶Ã‡Î“Ã¶Ã‡ DRAG TO ROTATE (mouse + touch) Î“Ã¶Ã‡Î“Ã¶Ã‡
            const onPointerDown = (e: PointerEvent) => {
                dragRef.current.isDragging = true;
                dragRef.current.prevX = e.clientX;
                dragRef.current.prevY = e.clientY;
                dragRef.current.velocityY = 0;
                dragRef.current.velocityX = 0;
                canvas.setPointerCapture(e.pointerId);
            };
            const onPointerMove = (e: PointerEvent) => {
                const dr = dragRef.current;
                if (!dr.isDragging) return;
                const dx = e.clientX - dr.prevX;
                const dy = e.clientY - dr.prevY;
                dr.velocityY = dx * 0.008;
                dr.velocityX = dy * 0.005;
                dr.rotY += dr.velocityY;
                dr.rotX += dr.velocityX;
                dr.rotX = Math.max(-1.2, Math.min(0.6, dr.rotX));
                dr.prevX = e.clientX;
                dr.prevY = e.clientY;
            };
            const onPointerUp = (e: PointerEvent) => {
                dragRef.current.isDragging = false;
                canvas.releasePointerCapture(e.pointerId);
            };
            canvas.addEventListener("pointerdown", onPointerDown);
            canvas.addEventListener("pointermove", onPointerMove);
            canvas.addEventListener("pointerup", onPointerUp);
            canvas.addEventListener("pointercancel", onPointerUp);

            // Resize
            const ro = new ResizeObserver(() => {
                if (!canvas || destroyed) return;
                renderer.setSize(canvas.clientWidth, canvas.clientHeight);
                camera.aspect = canvas.clientWidth / canvas.clientHeight;
                camera.updateProjectionMatrix();
            });
            ro.observe(canvas);

            // Store cleanup functions on the canvas object for the useEffect return
            (canvas as any)._cleanup = () => {
                canvas.removeEventListener("pointerdown", onPointerDown);
                canvas.removeEventListener("pointermove", onPointerMove);
                canvas.removeEventListener("pointerup", onPointerUp);
                canvas.removeEventListener("pointercancel", onPointerUp);
                ro.disconnect();
            };
        };

        load();
        canvas.addEventListener("click", handleClick);

        return () => {
            destroyed = true;
            observer.disconnect();
            if (animRef.current) cancelAnimationFrame(animRef.current);
            canvas.removeEventListener("click", handleClick);
            if ((canvas as any)._cleanup) (canvas as any)._cleanup();

            // Clean up 3D memory to optimize performance
            if (threeRef.current) {
                const { scene, renderer } = threeRef.current as any;
                if (scene) {
                    scene.traverse((obj: any) => {
                        if (obj.isMesh) {
                            if (obj.geometry) obj.geometry.dispose();
                            if (obj.material) {
                                if (Array.isArray(obj.material)) {
                                    obj.material.forEach((m: any) => m.dispose());
                                } else {
                                    obj.material.dispose();
                                }
                            }
                        }
                    });
                }
                if (renderer) renderer.dispose();
            }
        };
    }, [handleClick]);

    return <canvas ref={canvasRef} className="w-full h-full cursor-grab active:cursor-grabbing" style={{ display: "block", touchAction: "none" }} />;
}
